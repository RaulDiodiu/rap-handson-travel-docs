METHOD setbookingdate.
  " Read relevant booking instances
  READ ENTITIES OF zraph_##_i_travelwdtp IN LOCAL MODE
    ENTITY booking
      FIELDS ( bookingdate )
      WITH CORRESPONDING #( keys )
    RESULT DATA(bookings).

  " Loop over bookings and set system date as booking date
  LOOP AT bookings ASSIGNING FIELD-SYMBOL(<booking>) WHERE bookingdate IS INITIAL.
    <booking>-bookingdate = cl_abap_context_info=>get_system_date( ).
  ENDLOOP.

  " Save changes via EML
  MODIFY ENTITIES OF zraph_##_i_travelwdtp IN LOCAL MODE
    ENTITY booking
      UPDATE FIELDS ( bookingdate )
      WITH CORRESPONDING #( bookings )
    REPORTED DATA(reported_in).

  " Forward reported instances
  reported = CORRESPONDING #( DEEP reported_in ).
ENDMETHOD.

METHOD calculatetotalprice.
  READ ENTITIES OF zraph_##_i_travelwdtp IN LOCAL MODE
    ENTITY booking BY \_travel
      FROM CORRESPONDING #( keys )
    LINK DATA(links).

  MODIFY ENTITIES OF zraph_##_i_travelwdtp IN LOCAL MODE
    ENTITY travel
    EXECUTE recalctotalprice
    FROM VALUE #( FOR <link> IN links ( COND #( WHEN <link>-target-%is_draft = if_abap_behv=>mk-off THEN VALUE #( traveluuid = <link>-target-traveluuid ) ) ) )
    REPORTED DATA(reported_in).

  reported = CORRESPONDING #( DEEP reported_in ).
ENDMETHOD.

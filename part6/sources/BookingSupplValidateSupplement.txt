METHOD validatesupplement.
  " Read relevant booking supplement instance data
  READ ENTITIES OF zraph_##_i_travelwdtp IN LOCAL MODE
    ENTITY bookingsupplement
      FIELDS ( supplementid )
      WITH CORRESPONDING #( keys )
    RESULT DATA(supplements).

  READ ENTITIES OF zraph_##_i_travelwdtp IN LOCAL MODE
    ENTITY bookingsupplement BY \_travel
      FROM CORRESPONDING #( supplements )
    LINK DATA(links).

  " Get Supplement Master Data for comparison
  SELECT FROM /dmo/i_supplement AS md_supplement
    INNER JOIN @supplements AS supplement ON supplement~supplementid = md_supplement~supplementid
    FIELDS md_supplement~supplementid
    INTO TABLE @DATA(supplements_db).

  " Loop over all booking supplement instances to be saved
  LOOP AT supplements INTO DATA(supplement).
    " Invalidating State Messages from message table for state area booking supplement supplement validation
    APPEND VALUE #(  %tky               = supplement-%tky
                     %state_area        = 'VALIDATE_SUPPLEMENT' ) TO reported-bookingsupplement.

    " Raise messages for empty supplement ids
    IF supplement-supplementid IS  INITIAL.
      APPEND VALUE #( %tky = supplement-%tky ) TO failed-bookingsupplement.
      APPEND VALUE #( %tky                  = supplement-%tky
                      %state_area           = 'VALIDATE_SUPPLEMENT'
                      %msg                  = new_message( id       = 'ZRAPH_MSG_TRAVELWD'
                                                           number   = '012' " Supplement is initial
                                                           severity = if_abap_behv_message=>severity-error )
                      %path                 = VALUE #( travel-%tky = links[ KEY draft COMPONENTS source-%tky = supplement-%tky ]-target-%tky )
                      %element-supplementid = if_abap_behv=>mk-on ) TO reported-bookingsupplement.

    " Raise messages for non existing supplement ids
    ELSEIF supplement-supplementid IS NOT INITIAL AND NOT line_exists( supplements_db[ supplementid = supplement-supplementid ] ).
      APPEND VALUE #( %tky = supplement-%tky ) TO failed-bookingsupplement.
      APPEND VALUE #( %tky                  = supplement-%tky
                      %state_area           = 'VALIDATE_SUPPLEMENT'
                      %msg                  = new_message( id       = 'ZRAPH_MSG_TRAVELWD'
                                                           number   = '013' " Supplement &1 unknown
                                                           v1       = supplement-supplementid
                                                           severity = if_abap_behv_message=>severity-error )
                      %path                 = VALUE #( travel-%tky = links[ KEY draft COMPONENTS source-%tky = supplement-%tky ]-target-%tky )
                      %element-supplementid = if_abap_behv=>mk-on ) TO reported-bookingsupplement.
    ENDIF.
  ENDLOOP.
ENDMETHOD.

define view entity ZCDS_##_SPFLI
  as select from spfli as flight
  association [1..1] to scarr    as _Carrier     on _Carrier.carrid = flight.carrid
  association [1..1] to sairport as _AirportFrom on _AirportFrom.id = flight.airpfrom
  association [1..1] to sairport as _AirportTo   on _AirportTo.id = flight.airpto
{
  key carrid            as CarrierId,
  key connid            as ConnectionId,
      _Carrier.carrname as CarrierName,
      countryfr         as CountryFrom,
      cityfrom          as CityFrom,
      airpfrom          as AirportFrom,
      _AirportFrom.name as AirportFromName,
      countryto         as CountryTo,
      cityto            as CityTo,
      airpto            as AirportTo,
      _AirportTo.name   as AirportToName,
      fltime            as FlightTime,
      deptime           as DepartureTime,
      arrtime           as ArrivalTime,
      @Semantics.quantity.unitOfMeasure: 'DistanceUnit'
      distance          as DistanceAmount,
      distid            as DistanceUnit,
      fltype            as FlightType,
      period            as Period,

      _Carrier,
      _AirportFrom,
      _AirportTo
}
